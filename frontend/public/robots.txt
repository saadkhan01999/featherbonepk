# Feather & Bone — crawler rules
#
# The back office, the till and personal account pages are disallowed. This is
# NOT a security control — a Disallow line is a request, and anyone can ignore
# it. Those routes are protected by authentication; this only keeps them out of
# search results, where an indexed /admin login page is an invitation.
#
# The real reason it matters: pages behind a login return the app shell to a
# crawler, so a dozen "Feather & Bone" entries with no content end up competing
# with the menu pages that should be ranking.

User-agent: *

Disallow: /admin
Disallow: /pos
Disallow: /account
Disallow: /checkout
Disallow: /cart
Disallow: /wishlist
Disallow: /order/
Disallow: /reset-password
Disallow: /forgot-password
Disallow: /verify-email

# Search result pages: infinite variations of the same items, which is exactly
# what a crawl budget should not be spent on.
Disallow: /search?

Allow: /

# The API generates the sitemap from the live catalogue, so it is never stale —
# which is why it lives on the API's host, not here. Listing it in THIS file is
# what makes that legitimate: the sitemaps protocol accepts a sitemap on another
# host when the site's own robots.txt points to it, and every URL inside it is
# on featherbonepk.com (the API builds them from CLIENT_URL).
#
# In Search Console, add featherbonepk.com as a Domain property (DNS verified)
# so it covers the backend. subdomain too, then submit this same URL.
#
# `npm run package:webhoster` rewrites this host if the build targets a
# different API (e.g. --api-url for Railway), so it always matches the bundle.
Sitemap: https://backend.featherbonepk.com/sitemap.xml
